```bash
npm test -- --run src/features/plans/preview/PlanPreview.test.tsx
npm test -- --run src/features/plans/preview/PlanPreview.test.tsx src/features/method-form/MethodForm.test.tsx --reporter=default --reporter=junit --outputFile.junit=../artifacts/test-results/p5-05-junit.xml
npm test -- --run
npm run build
npm test -- --run --exclude src/features/scripts/ScriptRunsPage.test.tsx
```
