## 执行命令

以下命令从独立 V1 工作树的原型根目录执行。

```bash
set -o pipefail
audit=/home/mtd/liweigis/GeoPilot-round2/system-prototype-v1
evidence="$audit/artifacts/test-results/round2"
export GEOPILOT_QGIS_PROCESS="$audit/.environments/qgis-deb/qgis_process"
export PYTHONPATH="$audit/.environments/postgis-python"
export PYTHONDONTWRITEBYTECODE=1 PYTEST_DISABLE_PLUGIN_AUTOLOAD=1
"$audit/.venv/bin/python" -m pytest -q -p no:cacheprovider \
  tests/golden tests/gis tests/methods \
  tests/runners/test_measurement_steps.py \
  tests/application/test_plan_run.py::test_road_workflow_passes_fixed_road_cases_and_reproduces_in_an_independent_run \
  tests/llm/test_data_boundary.py \
  tests/llm/test_error_handling.py::test_actual_http_retry_posts_exact_scanned_prompt_with_fixed_envelope \
  tests/llm/test_error_handling.py::test_rejected_request_opens_no_socket_even_when_adapter_called_directly \
  tests/api/test_check_reports.py::test_real_qgis_execution_status_is_independent_of_check_report \
  --basetemp=artifacts/test-results/round2-fixed-cases-tmp \
  --junitxml="$evidence/gis-regression.xml" 2>&1 | tee "$evidence/gis-regression.log"
```
