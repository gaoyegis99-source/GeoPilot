```bash
set -o pipefail

npm ci --include=dev --no-audit --no-fund 2>&1 | tee "$frontend_evidence/frontend-npm-ci.log"

  --outputFile.junit="$frontend_evidence/frontend-vitest.xml" 2>&1 | tee "$frontend_evidence/frontend-vitest.log"
```
