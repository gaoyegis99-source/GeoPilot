## 正式工具命令

原始命令从原型根目录启动。两次 Docker 调用之间保留了调度交接窗口。DEM 退出 0，首次遥感退出 1。 [正式命令输出](p13-04-formal-tools.log) · [交接记录](p13-04-window.log)

```bash
set -o pipefail
scratch="$(mktemp -d "$PWD/artifacts/test-results/p13-04-formal-tools.XXXXXX")"
source_sha=a0c465d85d3a6b31fcd091ea4e913867bfc596df
image=sha256:c618d0d9d91d0cf473d61e5fc284aa335e87caabd51381b73cd61b2d0f93cb87
for family in raster remote-sensing; do
  container="geopilot-p13-04-${family}-tools-${scratch##*.}"
  docker run --name "$container" --rm --network none --user "$(id -u):$(id -g)" \
    --entrypoint /runtime/venv/bin/python --workdir /project \
    -e HOME=/tmp -e QT_QPA_PLATFORM=offscreen -e PYTHONDONTWRITEBYTECODE=1 \
    -e OPENBLAS_NUM_THREADS=1 -e GDAL_NUM_THREADS=1 -e OMP_NUM_THREADS=1 \
    --mount "type=bind,src=$PWD,dst=/project,readonly" \
    --mount "type=bind,src=$PWD/.environments/p13-native,dst=/runtime,readonly" \
    --mount "type=bind,src=$PWD/.environments/qgis-deb/usr/bin/qgis_process,dst=/usr/bin/qgis_process,readonly" \
    --mount "type=bind,src=$PWD/.environments/qgis-deb/usr/bin/qgis_process.bin,dst=/usr/bin/qgis_process.bin,readonly" \
    --mount "type=bind,src=$scratch,dst=/scratch" "$image" \
    scripts/bench/run_raster.py --profile "benchmarks/datasets/$family.yaml" --tiers S M L --workload tools \
    --source-commit "$source_sha" --output "/scratch/$family"
  code=$?
  printf 'end_family %s exit=%s utc=%s\n' "$family" "$code" "$(date -u +%FT%TZ)"
  if [ "$code" -ne 0 ]; then
    exit "$code"
  fi
done
```
