从原型目录执行。

```bash
export PYTHONPATH=src PYTHONDONTWRITEBYTECODE=1
export GEOPILOT_QGIS_PROCESS="$PWD/.environments/qgis-deb/qgis_process"
.venv/bin/python -m geopilot.diagnostics --json > docs/round2/deployment/p16-01-doctor.json
doctor_probe_dir=$(mktemp -d /tmp/geopilot-missing-qgis.XXXXXX)
.venv/bin/python -m geopilot.diagnostics --json --qgis-process "$doctor_probe_dir/missing-qgis" \
  > docs/round2/deployment/p16-01-missing-qgis.json

set -o pipefail
PYTEST_DISABLE_PLUGIN_AUTOLOAD=1 .venv/bin/python -m pytest -q -p no:cacheprovider tests/diagnostics \
  --junitxml=docs/round2/deployment/p16-01-junit.xml 2>&1 | tee docs/round2/deployment/p16-01-pytest.log
```
