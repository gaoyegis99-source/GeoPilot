## 实际命令

从原型目录运行。

```bash
set -o pipefail
.venv/bin/python scripts/methods/method_cli.py template --tool qgis-buffer@1 --method buffer-review@1 \
  > docs/round2/methods/p14-02-example.json
PYTHONDONTWRITEBYTECODE=1 PYTHONPATH=src python -c 'from pathlib import Path
import json
from jsonschema import Draft202012Validator
from geopilot.domain.tool_contract import ToolContract
from geopilot.methods.models import MethodDefinition
payload=json.loads(Path("docs/round2/methods/p14-02-example.json").read_text(encoding="utf-8"))
assert payload["status"]=="ok" and payload["data"]["scope"]=="method_template"
data=payload["data"]
contracts=(("ToolContract",ToolContract,ToolContract.model_json_schema(),data["tool"]),("MethodDefinition",MethodDefinition,json.loads(Path("artifacts/schemas/method-definition.schema.json").read_text(encoding="utf-8")),data["method"]))
for name,model,schema,document in contracts:
 Draft202012Validator.check_schema(schema)
 Draft202012Validator(schema).validate(document)
 model.model_validate(document)
 print(f"{name}: Draft 2020-12 and Pydantic PASS")
assert data["method"]["steps"][0]["tool"]=={"id":"qgis-buffer","version":1}
assert len(data["todos"])==6
assert all("default" not in option for option in data["method"]["critical_options"].values())
print("Exact qgis-buffer@1 and six unfilled required values: PASS")' | tee docs/round2/methods/p14-02-schema.log

base=$(mktemp -d /tmp/geopilot-p14-02-template.XXXXXX)
PYTEST_DISABLE_PLUGIN_AUTOLOAD=1 PYTHONDONTWRITEBYTECODE=1 .venv/bin/python -m pytest \
  tests/method_dev/test_template_generation.py -q -p no:cacheprovider --basetemp="$base" \
  --junitxml=docs/round2/methods/p14-02-template-junit.xml 2>&1 | tee docs/round2/methods/p14-02-template.log

base=$(mktemp -d /tmp/geopilot-p14-02-cli.XXXXXX)
PYTEST_DISABLE_PLUGIN_AUTOLOAD=1 PYTHONDONTWRITEBYTECODE=1 .venv/bin/python -m pytest \
  tests/method_dev/test_method_cli.py -q -p no:cacheprovider --basetemp="$base" \
  --junitxml=docs/round2/methods/p14-02-cli-regression-junit.xml 2>&1 | tee docs/round2/methods/p14-02-cli-regression.log

base=$(mktemp -d /tmp/geopilot-p14-02-review.XXXXXXXX)
PYTEST_DISABLE_PLUGIN_AUTOLOAD=1 PYTHONDONTWRITEBYTECODE=1 .venv/bin/python -m pytest \
  tests/method_dev/test_template_generation.py tests/method_dev/test_method_cli.py \
  -q -p no:cacheprovider --basetemp="$base" \
  --junitxml=docs/round2/methods/p14-02-review-junit.xml 2>&1 | tee docs/round2/methods/p14-02-review.log
```
