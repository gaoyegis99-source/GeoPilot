```bash
set -euo pipefail
export PYTHONDONTWRITEBYTECODE=1
dem_regen="$(mktemp -d "$PWD/artifacts/test-results/p14-04-dem-regen.XXXXXX")"
{
  .venv/bin/python tests/method_dev/fixtures/dem-terrain/generate_inputs.py --output "$dem_regen"
  sha256sum tests/fixtures/raster/dem-plane.tif
  for target in dem.tif unlabelled-dem.tif; do
    cmp "$dem_regen/$target" "tests/method_dev/fixtures/dem-terrain/inputs/$target"
    sha256sum "tests/method_dev/fixtures/dem-terrain/inputs/$target"
  done
  if .venv/bin/python tests/method_dev/fixtures/dem-terrain/generate_inputs.py --output "$dem_regen"; then
    exit 1
  else

    printf 'existing output exit=%s\n' "$result"

  fi
} 2>&1 | tee docs/round2/methods/p14-04-dem-regen.log
```
