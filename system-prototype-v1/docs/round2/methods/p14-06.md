    GEOPILOT_QGIS_PROCESS=/nonexistent/qgis-process \
      .venv/bin/python scripts/methods/check_versions.py --baseline . --candidate .
    PYTEST_DISABLE_PLUGIN_AUTOLOAD=1 PYTHONDONTWRITEBYTECODE=1 \
      .venv/bin/python -m pytest -q -p no:cacheprovider \
      tests/method_dev/test_version_compatibility.py tests/plans/test_plan_versioning.py \
      tests/remote_sensing/test_config_registry.py tests/methods/test_road_distance_modes.py \
      --junitxml=docs/round2/methods/p14-06-junit.xml
    PYTEST_DISABLE_PLUGIN_AUTOLOAD=1 PYTHONDONTWRITEBYTECODE=1 \
      GEOPILOT_QGIS_PROCESS="$PWD/.environments/qgis-deb/qgis_process" \
